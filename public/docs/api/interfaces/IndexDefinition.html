    Interface IndexDefinition<TDocument>

    How to build an index, passed to SinterCollection.createIndex.

    interface IndexDefinition<TDocument extends object = Document> {
        direction?: 1 | -1;
        field: Exclude<
            TDocument extends object
                ? {
                    [Key in string]: | Key
                    | (
                        NonNullable<TDocument[Key]> extends AtomicValue | readonly unknown[]
                            ? never
                            : NonNullable<TDocument[Key]> extends object
                                ? `${Key}.${NonNullable<(...)[(...)]> extends object
                                    ? { [Key in (...)]: (...) }[(...) & (...)]
                                    : never}`
                                : never
                    )
                }[keyof TDocument & string]
                : never,
        >;
        name?: string;
        sparse?: boolean;
        unique?: boolean;
    }
